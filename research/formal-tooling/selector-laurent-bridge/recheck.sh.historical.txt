#!/usr/bin/env bash
set -euo pipefail
HERE="$(cd "$(dirname "$0")" && pwd)"
UPSTREAM=${ARTIFACT_ROOT}/vermilion-tooling/vermilion
source ${ARTIFACT_ROOT}/vermilion-tooling/env.sh
export LEAN_PATH="$HERE:${LEAN_PATH:-}"
cd "$UPSTREAM"
cmp "$HERE/ResidueBridge.lean" ${ARTIFACT_ROOT}/lean-koszul/ResidueBridge.lean
for module in Evidence residue_term missing_face_zero; do
  cmp "$HERE/residue/$module.lean" "$UPSTREAM/examples/residue-selector/proofs/residue/$module.lean"
done
for module in ResidueBridge residue/Evidence residue/residue_term residue/missing_face_zero SelectorLaurentBridge; do
  lake env lean "$HERE/$module.lean" -R "$HERE" -o "$HERE/$module.olean"
done
